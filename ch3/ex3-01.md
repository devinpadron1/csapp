## Practice Problem 3.1
%rax    -> 0x100

0x104   -> 0xAB

$0x108  -> 0x108

(%rax)  -> 0xFF 
    (rax) == M[R[rax]]
    1. R[rax] -> 0x100
    2. M[0x100] -> 0XFF

4(%rax) -> 0xAB
    Imm(rax) -> M[Imm + R[rax]]
    1. M[4 + 0x100] -> M[0x104]
    2. M[0x104] -> 0xAB

9(%rax,%rdx) -> 0x11
    Imm(rb,ri) -> M[Imm + R[rax] + R[rdx]] 
    1. M[9 + 0x100 + 0x3]
    2. M[10C]
    3. 0x11

260(%rcx,%rdx) -> 0x13 
    0x1 + 0x3 == 0x4
    260 + 0x4 = 264(decimal)

    convert 0xFF to decimal
        15*16^1 + 15*16^0
        240 + 15 = 255
        
        therefore
        0x108 => 264

0xFC(,%rcx,4) -> 0xFF 
    M[Imm + R[rcx] * 4] == M[0xFC + 0x4]
    M[0x100] -> 0xFF

(%rax,%rdx,4) -> 0x11
